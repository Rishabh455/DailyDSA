class Solution {
    public boolean solve(String s,int i, int minJump,int maxJump){
                int n=s.length();
                if(i==n-1)return true;
                for(int jump= minJump;jump<=maxJump;jump++){
                    int idx=i+jump;
                    if(idx>=n)break;
                    if(s.charAt(idx)=='0'){
                    if(    solve(s,idx,minJump,maxJump)==true)return true;
                    }
                }
                return false;
    }
    public boolean canReach(String s, int minJump, int maxJump) {
      //  int n=s.length();
       return solve( s, 0,minJump,maxJump);
        
    }
}

class Solution {
    
    public boolean canReach(String s, int minJump, int maxJump) {
       ArrayList<Integer> dp = new ArrayList<>(Collections.nCopies(s.length(), 0));
       int n=s.length();
       dp.set(0,1);
       int count=0;
       for(int i=1;i<n;i++){
            int a=i-minJump;
            if(a>=0){
                count+=dp.get(a);
            }
            int b=i-maxJump-1;
            if(b>=0){
                count-=dp.get(b);
            }
            if(count>0 && s.charAt(i)=='0'){
                dp.set(i,1);
            }
       }
       
      //  int n=s.length();
       return dp.get(n-1)>0;
        
    }
}
